            return;

        if (matchIDs !== '') {
            $.ajax({
                url: '/iframe/livescores/get-matches-info?match_ids=' + matchIDs,

                dataType: 'json',
                success: function (response) {
                    $.each(response.data, function (index, match) {
                        $('#match-card-' + match.match_id).data('livescore-state', match.livescore_state)

                        if (match.has_livescore_started) {
                            $('#matchTime_' + match.match_id).show();
                        } else {
                            $('#matchTime_' + match.match_id).hide();
                        }

                        $('#match_current_period_' + match.match_id).html(match.current_period);
                        $('#match_minute_' + match.match_id).html(match.minute);
                        $('#match_second_' + match.match_id).html(match.second);

                        if ($('#home_team_result_' + match.match_id).html() != match.home_team_result) {
                            $('#home_team_result_' + match.match_id).html(match.home_team_result);
                            $('#home_team_result_' + match.match_id).addClass('changed');
                            setTimeout(function () {
                                $('#home_team_result_' + match.match_id).removeClass('changed');
                            }, 2000);
                        }
                        if ($('#visiting_team_result_' + match.match_id).html() != match.visiting_team_result) {
                            $('#visiting_team_result_' + match.match_id).html(match.visiting_team_result);

                            setTimeout(function () {
                                $('#visiting_team_result_' + match.match_id).removeClass('changed');

                        }

                },

                }

            if (checkIfLivescoreHasStartedOrIsAboutToStart($(this))) {
                matchIDs += ',' + $(this).data('match-id');

    function checkIfLivescoreHasStartedOrIsAboutToStart(matchCard) {
        if (matchCard.data('livescore-state') == 'stopped') {
            return false;
        }

        if (matchCard.data('livescore-state') == 'started') {
            return true;
        }

        if (matchCard.data('match-date') != "2026-11-08") {
            return false;
        }

        var inputTime = moment(matchCard.data('match-time'), 'HH:mm:ss');
        var currentTime = moment();
        var minutesDifference = inputTime.diff(currentTime, 'minutes');
        if (inputTime.diff(currentTime, 'minutes') > 0 && Math.abs(minutesDifference) > 15) {
            return false
        }

        return true;
    }
